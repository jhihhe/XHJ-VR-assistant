              //获取当前时间并打印

              let mm =
                new Date().getMonth() + 1 < 10
                  ? "0" + (new Date().getMonth() + 1)
                  : new Date().getMonth() + 1;
              let dd =
                new Date().getDate() < 10

                new Date().getHours() < 10

                  : new Date().getMinutes();

                new Date().getSeconds() < 10
                  ? "0" + new Date().getSeconds()
                  : new Date().getSeconds();
              return ischeck
                ? (_this.gettime =
                    yy + "-" + mm + "-" + dd + " " + hh + ":" + mf + ":" + ss)
